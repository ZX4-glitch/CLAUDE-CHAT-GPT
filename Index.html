export default { async fetch(request, env) { const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }; const json = (data, status = 200) => new Response(JSON.stringify(data), { status, headers: { ...cors, "Content-Type": "application/json; charset=utf-8", }, }); if (request.method === "OPTIONS") { return new Response(null, { headers: cors }); } const url = new URL(request.url); if (url.pathname !== "/api/chat" || request.method !== "POST") { return json({ error: "Ruta no encontrada." }, 404); } let body; try { body = await request.json(); } catch { return json({ error: "El mensaje no es válido." }, 400); } const message = body.message; if ( typeof message !== "string" || !message.trim() || message.length > 4000 ) { return json({ error: "Escribe un mensaje de hasta 4000 caracteres." }, 400); } const [gptResult, claudeResult] = await Promise.allSettled([ fetch("https://api.openai.com/v1/responses", { method: "POST", headers: { "Authorization": `Bearer ${env.OPENAI_API_KEY}`, "Content-Type": "application/json", }, body: JSON.stringify({ model: "gpt-4.1-mini", input: message.trim(), max_output_tokens: 700, }), }).then(async r => { const data = await r.json(); if (!r.ok) throw new Error("OpenAI: " + (data.error?.message || r.status)); return data.output ?.flatMap(item => item.content || []) .filter(item => item.type === "output_text") .map(item => item.text) .join("\n") || "Sin respuesta de ChatGPT."; }), fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { "x-api-key": env.ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01", "Content-Type": "application/json", }, body: JSON.stringify({ model: "claude-sonnet-4-6", max_tokens: 700, messages: [ { role: "user", content: message.trim() } ], }), }).then(async r => { const data = await r.json(); if (!r.ok) throw new Error("Claude: " + (data.error?.message || r.status)); return data.content ?.filter(item => item.type === "text") .map(item => item.text) .join("\n") || "Sin respuesta de Claude."; }), ]); const result = { gpt: gptResult.status === "fulfilled" ? gptResult.value : "Error: " + gptResult.reason.message, claude: claudeResult.status === "fulfilled" ? claudeResult.value : "Error: " + claudeResult.reason.message, }; return json(result); }, };
